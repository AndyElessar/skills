# Incremental Generator Template

Use this as a starting point for attribute-driven generators. Rename placeholders and extract a repository-specific model before adding features.

```csharp
using System.Collections.Immutable;
using System.Text;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using Microsoft.CodeAnalysis.Text;

namespace MyGeneratorPackage;

[Generator]
public sealed class MyGenerator : IIncrementalGenerator
{
    private const string AttributeMetadataName = "MyGeneratorPackage.MyAttribute";

    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        context.RegisterPostInitializationOutput(static context =>
        {
            context.AddSource("MyAttribute.g.cs", SourceText.From(AttributeSource, Encoding.UTF8));
        });

        IncrementalValuesProvider<MyModel> models = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                AttributeMetadataName,
                predicate: static (node, _) => node is TypeDeclarationSyntax,
                transform: static (context, cancellationToken) => CreateModel(context, cancellationToken))
            .Where(static model => model is not null)
            .Select(static (model, _) => model!.Value)
            .WithTrackingName("MyGenerator.Models");

        IncrementalValueProvider<ImmutableArray<MyModel>> collectedModels = models
            .Collect()
            .Select(static (models, _) => models.Sort(MyModelComparer.Instance));

        context.RegisterSourceOutput(collectedModels, static (context, models) => Emit(context, models));
    }

    private static MyModel? CreateModel(GeneratorAttributeSyntaxContext context, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();

        if (context.TargetSymbol is not INamedTypeSymbol typeSymbol)
        {
            return null;
        }

        string typeName = typeSymbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);
        string hintName = typeName.Replace("global::", string.Empty).Replace('.', '_').Replace('<', '_').Replace('>', '_');
        return new MyModel(typeName, hintName);
    }

    private static void Emit(SourceProductionContext context, ImmutableArray<MyModel> models)
    {
        foreach (MyModel model in models)
        {
            string source = $$"""
                // <auto-generated/>
                #nullable enable

                namespace MyGeneratorPackage.Generated;

                internal static partial class {{model.HintName}}Generated
                {
                    public const string TargetType = "{{model.TypeName}}";
                }
                """;

            context.AddSource($"{model.HintName}.g.cs", SourceText.From(source, Encoding.UTF8));
        }
    }

    private const string AttributeSource = """
        // <auto-generated/>
        #nullable enable

        namespace MyGeneratorPackage;

        [global::System.AttributeUsage(global::System.AttributeTargets.Class | global::System.AttributeTargets.Struct, AllowMultiple = false)]
        internal sealed class MyAttribute : global::System.Attribute
        {
        }
        """;

    private readonly record struct MyModel(string TypeName, string HintName);

    private sealed class MyModelComparer : IComparer<MyModel>
    {
        public static readonly MyModelComparer Instance = new();

        public int Compare(MyModel left, MyModel right)
            => StringComparer.Ordinal.Compare(left.HintName, right.HintName);
    }
}
```

## Adaptation Notes

- Use `ForAttributeWithMetadataName` when the minimum supported Roslyn version allows it. Use `CreateSyntaxProvider` only for non-attribute triggers or older SDK support.
- Replace `MyModel` with a small equatable model. Do not pass `SyntaxNode`, `ISymbol`, `SemanticModel`, `Compilation`, or `Location` downstream.
- Add diagnostics by returning a cacheable result model and reporting diagnostics at the output edge.
- Add `WithTrackingName` to stages that should be tested for cacheability.
- Sort before emission so hint names and snapshots are stable.
